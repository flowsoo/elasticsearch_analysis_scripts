sudo -u elastic sh -c '
  PASS=$(/usr/share/elasticsearch/bin/elasticsearch-keystore show \
    xpack.security.transport.ssl.keystore.secure_password)
  keytool -list -storetype PKCS12 \
    -keystore /etc/elasticsearch/certs/your-node.p12 \
    -storepass "$PASS" >/dev/null
'

  Check the exit code with:

echo $?
